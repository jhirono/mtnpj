Build the Cloudflare Worker API (Hono) that the Pages frontend will consume. The Worker must expose REST endpoints for routes and areas with server-side SQL filtering, FTS5 text search, pagination, and parameterized queries against the D1 schema produced by Plan 02.     2. **`worker-api/src/db/queries.ts`** — SQL builders. CRITICAL: only the whitelist-validated `type` enum can be interpolated as a column name; all other params are parameterized:

       ```typescript
       import { ALLOWED_TYPE_COLUMNS, AllowedType } from '../types';

       export interface RouteFilters {
         q?: string;
         grade?: string;
         grade_min?: number;
         grade_max?: number;
         type?: AllowedType;
         region?: string;
         stars_min?: number;
         votes_min?: number;
         area_id?: string;
         page: number;
         limit: number;
       }

       export function buildRoutesQuery(f: RouteFilters): { sql: string; params: unknown[] } {
         const conds: string[] = ['1=1'];
         const params: unknown[] = [];

         // FTS5 full-text search
         if (f.q) {
           conds.push(`r.rowid IN (SELECT rowid FROM routes_fts WHERE routes_fts MATCH ?)`);
           params.push(f.q);
         }
         if (f.grade) { conds.push('r.route_grade = ?'); params.push(f.grade); }
         if (f.grade_min !== undefined) { conds.push('r.route_grade_numeric >= ?'); params.push(f.grade_min); }
         if (f.grade_max !== undefined) { conds.push('r.route_grade_numeric <= ?'); params.push(f.grade_max); }
         if (f.type) {
           // type is enum-validated; column name is sourced from ALLOWED_TYPE_COLUMNS,
           // never from raw user input. Static-known column name = safe to interpolate.
           const col = ALLOWED_TYPE_COLUMNS[f.type];
           conds.push(`r.${col} = 1`);
         }
         if (f.region) { conds.push('a.path LIKE ?'); params.push(`/${f.region}/%`); }
         if (f.stars_min !== undefined) { conds.push('r.route_stars >= ?'); params.push(f.stars_min); }
         if (f.votes_min !== undefined) { conds.push('r.route_votes >= ?'); params.push(f.votes_min); }
         if (f.area_id) {
           // descendants via materialized path: include the area itself and all under its path prefix
           conds.push(`(r.area_id = ? OR a.path LIKE (SELECT path FROM areas WHERE area_id = ?) || '%')`);
           params.push(f.area_id, f.area_id);
         }

         const offset = (f.page - 1) * f.limit;
         const sql = `
           SELECT r.*, a.path AS area_path, a.area_name AS area_name
           FROM routes r
           JOIN areas a ON r.area_id = a.area_id
           WHERE ${conds.join(' AND ')}
           ORDER BY r.route_votes DESC, r.route_stars DESC
           LIMIT ? OFFSET ?
         `;
         params.push(f.limit, offset);
         return { sql, params };
       }

       export function buildAreasQuery(opts: {
         q?: string; region?: string; parent_id?: string; page: number; limit: number;
       }): { sql: string; params: unknown[] } {
         const conds: string[] = ['1=1'];
         const params: unknown[] = [];
         if (opts.q) { conds.push('area_name LIKE ?'); params.push(`%${opts.q}%`); }
         if (opts.region) { conds.push('path LIKE ?'); params.push(`/${opts.region}/%`); }
         if (opts.parent_id) { conds.push('parent_id = ?'); params.push(opts.parent_id); }
         const offset = (opts.page - 1) * opts.limit;
         const sql = `SELECT * FROM areas WHERE ${conds.join(' AND ')} ORDER BY area_name LIMIT ? OFFSET ?`;
         params.push(opts.limit, offset);
         return { sql, params };
       }

       import { Hono } from 'hono';
       import { zValidator } from '@hono/zod-validator';
       import { z } from 'zod';
       import type { Bindings, RouteRow } from '../types';
       import { ALLOWED_TYPES } from '../types';
       import { buildRoutesQuery } from '../db/queries';

       const app = new Hono<{ Bindings: Bindings }>();

       const routeQuerySchema = z.object({
         q: z.string().min(1).optional(),
         grade: z.string().optional(),
         grade_min: z.coerce.number().optional(),
         grade_max: z.coerce.number().optional(),
         type: z.enum(ALLOWED_TYPES).optional(),
         region: z.string().regex(/^[a-z0-9-]+$/, 'region must be slug').optional(),
         stars_min: z.coerce.number().min(0).max(4).optional(),
         votes_min: z.coerce.number().min(0).optional(),
         area_id: z.string().optional(),
         page: z.coerce.number().int().min(1).default(1),
         limit: z.coerce.number().int().min(1).max(200).default(50),
       });

       app.get('/routes', zValidator('query', routeQuerySchema), async (c) => {
         const f = c.req.valid('query');
         const { sql, params } = buildRoutesQuery(f);
         const { results } = await c.env.DB.prepare(sql).bind(...params).all<RouteRow & { area_path: string; area_name: string }>();
         return c.json({ data: results ?? [], page: f.page, limit: f.limit });
       });

       app.get('/routes/:id', async (c) => {
         const id = c.req.param('id');
         const row = await c.env.DB
           .prepare('SELECT r.*, a.path AS area_path, a.area_name FROM routes r JOIN areas a ON r.area_id = a.area_id WHERE r.route_id = ? LIMIT 1')
           .bind(id)
           .first<RouteRow & { area_path: string; area_name: string }>();
         if (!row) return c.json({ error: 'route not found' }, 404);
         return c.json({ data: row });
       });

       export default app;
       ```

|----------|-------------|
| Public internet → Worker `/api/*` | Untrusted query parameters (q, grade, type, region, etc.) and path params (route id, area id) |
| Worker → D1 | All values bound via `?` placeholders except whitelisted boolean column names |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-01 | Tampering / SQLi | `?type=` parameter (column-name interpolation) | mitigate | `z.enum(ALLOWED_TYPES)` rejects values outside the 9-item whitelist with 400. Column name resolved via `ALLOWED_TYPE_COLUMNS[f.type]` map lookup; never raw user input. Tested in routes.test.ts SQL-injection test. |
| T-03-02 | Tampering / SQLi | All other query/path params | mitigate | Every other value is bound with `.bind(...)` parameterized placeholder; zod coerces numeric params; `region` is regex-restricted to `^[a-z0-9-]+$`. |
| T-03-03 | Information Disclosure | error responses leaking SQL | mitigate | `app.onError` returns `{error: 'internal error'}` — internal exception detail not surfaced; logged via console.error. |
| T-03-04 | Tampering | CORS open to all origins | mitigate | `cors({ origin: ... })` allows only `localhost:5173` and `*.pages.dev`; other origins receive no `Access-Control-Allow-Origin`. |
| T-03-05 | DoS | unbounded `limit` causing huge result sets | mitigate | `z.coerce.number().min(1).max(200)` caps at 200; oversized requests return 400. |
| T-03-06 | Spoofing | none — public read API | accept | No authentication; no authorized endpoints; D-09 confirms public read-only access is acceptable. |
| T-03-07 | DoS | API abuse / scraping the scraper | accept | Cloudflare WAF/rate limiting at the edge handles this (per RESEARCH security-domain table); not an app-level concern in Phase 1. |
