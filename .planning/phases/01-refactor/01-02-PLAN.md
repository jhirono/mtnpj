       def parse_route_type_to_booleans(route_type):
           result = {col: 0 for col in BOOL_COLUMNS}
           if not route_type:
               return result
           # Normalize: pad with spaces so " tr " keyword can match boundaries
           lower = " " + route_type.lower() + " "

           # (we don't want "Grade III" to false-match anything; we keep them out by skipping)

           return result

       def build_materialized_path(area_hierarchy):
           segments = []
           for h in area_hierarchy or []:
               url = (h.get("area_hierarchy_url") or "").rstrip("/")
               if not url:
                   continue
               last = url.split("/")[-1]
               # Skip the All-Locations stub
               if last in ("route-guide", ""):

               # If last segment is purely numeric (only "/area/NNN/" with no slug), skip

               segments.append(last)

    4. **`grade_to_numeric(grade: str | None) -> float | None`**:
       ```python
       def grade_to_numeric(grade):
           if not grade or grade == "Unknown":
               return None
           m = re.match(r"^5\.(\d+)([a-d]?)([+\-]?)$", grade.strip())
           if not m:
               return None  # boulder grades (V0…V16) and other formats unsupported here
           minor = int(m.group(1))
           letter = m.group(2)
           # 5.9 = 9.0; 5.10a = 10.1; 5.10b = 10.2; 5.10c = 10.3; 5.10d = 10.4
           letter_offset = {"": 0.0, "a": 0.1, "b": 0.2, "c": 0.3, "d": 0.4}.get(letter, 0.0)
           return float(minor) + letter_offset
       ```

       def extract_mp_id_from_url(url):
           m = re.search(r"/(?:area|route)/(\d+)/", url or "")
           if m:
               return m.group(1)
           # Deterministic fallback for malformed URLs

    6. **`sql_quote(value) -> str`** (SQL-standard escape):
       ```python
       def sql_quote(value):
           if value is None:
               return "NULL"
           if isinstance(value, bool):
               return "1" if value else "0"
           if isinstance(value, (int, float)):
               return str(value)
           # Escape single quotes by doubling them (SQL standard)
           s = str(value).replace("'", "''")
           return f"'{s}'"
       ```
